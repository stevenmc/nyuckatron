# cryptography (a google-auth dependency) is pinned old ONLY on Intel macOS:
# 49+ dropped macOS x86_64/universal2 wheels (arm64-only now), and there's no
# working Xcode Command Line Tools on that kind of dev machine to compile
# from source instead -- confirmed pip otherwise picks the newest version by
# version number first and tries to build THAT from source, rather than
# falling back to an older version with an available wheel, so an unpinned
# or open-ended constraint doesn't fix this on its own. Linux (the actual
# EC2 deployment target) has full wheel coverage for every recent version
# on every common architecture (x86_64, aarch64, ...) -- confirmed via
# PyPI's file listing -- so it isn't held back by this and gets the latest
# compatible version instead.
cryptography==48.0.1; sys_platform == "darwin" and platform_machine == "x86_64"
cryptography>=42.0.3; sys_platform != "darwin" or platform_machine != "x86_64"
